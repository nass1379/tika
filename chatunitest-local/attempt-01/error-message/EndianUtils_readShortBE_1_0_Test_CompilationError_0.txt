package org.apache.tika.io;

import org.apache.tika.io.EndianUtils;
import org.apache.tika.exception.TikaException;
import org.apache.tika.io.BufferUnderrunException;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.InputStream;
import java.io.IOException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class EndianUtils_readShortBE_1_0_Test {

    @Mock
    private InputStream inputStream;

    @Test
    public void testReadShortBE() throws IOException, TikaException {
        when(inputStream.read()).thenReturn(0x12, 0x34);
        short result = EndianUtils.readShortBE(inputStream);
        assertEquals(0x1234, result);
        verify(inputStream, times(2)).read();
    }

    @Test
    public void testReadShortBEWithBufferUnderrun() throws IOException, TikaException {
        when(inputStream.read()).thenReturn(-1);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readShortBE(inputStream));
        verify(inputStream, times(1)).read();
    }
}

package org.apache.tika.io;

import org.apache.tika.exception.TikaException;
import org.apache.tika.io.EndianUtils;
import org.apache.tika.io.InputStreams;
import org.apache.tika.io.BufferUnderrunException;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class EndianUtils_readUShortLE_2_0_Test {

    @Mock
    private InputStream mockInputStream;

    @Test
    public void testReadUShortLE() throws IOException, TikaException {
        // Arrange
        int ch1 = 0x12;
        int ch2 = 0x34;
        when(mockInputStream.read()).thenReturn(ch1, ch2);
        // Act
        int result = EndianUtils.readUShortLE(mockInputStream);
        // Assert
        assertEquals((ch2 << 8) + (ch1), result);
        verify(mockInputStream, times(2)).read();
    }

    @Test
    public void testReadUShortLE_ThrowsBufferUnderrunException() throws IOException, TikaException {
        // Arrange
        when(mockInputStream.read()).thenReturn(-1);
        // Act & Assert
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readUShortLE(mockInputStream));
        verify(mockInputStream, times(1)).read();
    }
}

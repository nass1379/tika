public class EndianUtils_readLongLE_9_0_Test {

    @Test
    public void testReadLongLE() throws IOException {
        byte[] data = ByteBuffer.allocate(8).putLong(0x123456789ABCDEF0L).array();
        InputStream inputStream = new ByteArrayInputStream(data);
        long result = EndianUtils.readLongLE(inputStream);
        assertEquals(0x123456789ABCDEF0L, result);
    }

    @Test
    public void testReadLongLE_WithBufferUnderrun() throws IOException {
        byte[] data = new byte[0];
        InputStream inputStream = new ByteArrayInputStream(data);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readLongLE(inputStream));
    }

    @Test
    public void testReadLongLE_WithNegativeByte() throws IOException {
        byte[] data = ByteBuffer.allocate(8).putLong(-1L).array();
        InputStream inputStream = new ByteArrayInputStream(data);
        long result = EndianUtils.readLongLE(inputStream);
        assertEquals(-1L, result);
    }
}

public class EndianUtils_readUShortBE_3_0_Test {

    @Test
    public void testReadUShortBE_Success() throws IOException {
        byte[] data = { 0x12, 0x34 };
        InputStream inputStream = new ByteArrayInputStream(data);
        int result = EndianUtils.readUShortBE(inputStream);
        assertEquals(0x1234, result);
    }

    @Test
    public void testReadUShortBE_InsufficientData() throws IOException {
        byte[] data = { 0x12 };
        InputStream inputStream = new ByteArrayInputStream(data);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readUShortBE(inputStream));
    }

    @Test
    public void testReadUShortBE_NegativeData() throws IOException {
        byte[] data = { 0xFF, 0xFF };
        InputStream inputStream = new ByteArrayInputStream(data);
        int result = EndianUtils.readUShortBE(inputStream);
        assertEquals(0xFFFF, result);
    }
}

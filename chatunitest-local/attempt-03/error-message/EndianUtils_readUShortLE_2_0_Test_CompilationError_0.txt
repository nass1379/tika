public class EndianUtils_readUShortLE_2_0_Test {

    @Test
    public void testReadUShortLE() throws IOException {
        byte[] data = { 0x12, 0x34 };
        InputStream inputStream = new ByteArrayInputStream(data);
        int result = EndianUtils.readUShortLE(inputStream);
        assertEquals(0x3412, result);
    }

    @Test
    public void testReadUShortLE_WithBufferUnderrun() throws IOException {
        byte[] data = { 0x12 };
        InputStream inputStream = new ByteArrayInputStream(data);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readUShortLE(inputStream));
    }

    @Test
    public void testReadUShortLE_WithNegativeValues() throws IOException {

        int result = EndianUtils.readUShortLE(inputStream);
        assertEquals(0xFEFF, result);
    }

public class EndianUtils_readUE7_11_0_Test {

    @Test
    public void testReadUE7_Success() throws IOException {
        byte[] data = { 0x01, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00 };
        InputStream inputStream = new ByteArrayInputStream(data);
        long result = EndianUtils.readUE7(inputStream);
        assertEquals(1, result);
    }

    @Test
    public void testReadUE7_MultipleBytes() throws IOException {
        byte[] data = { 0x81, 0x80, 0x80, 0x80, 0x80, 0x80, 0x80, 0x80, 0x00 };
        InputStream inputStream = new ByteArrayInputStream(data);
        long result = EndianUtils.readUE7(inputStream);
        assertEquals(127, result);
    }

    @Test
    public void testReadUE7_MaxValue() throws IOException {
        byte[] data = { 0x7F, 0x7F, 0x7F, 0x7F, 0x7F, 0x7F, 0x7F, 0x7F, 0x7F };
        InputStream inputStream = new ByteArrayInputStream(data);
        long result = EndianUtils.readUE7(inputStream);
        assertEquals(127 + (127 << 7) + (127 << 14) + (127 << 21) + (127 << 28) + (127 << 35) + (127 << 42) + (127 << 49), result);
    }

    @Test
    public void testReadUE7_BufferUnderrun() {
        InputStream inputStream = new ByteArrayInputStream(new byte[0]);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readUE7(inputStream));
    }

    @Test
    public void testReadUE7_NegativeByte() throws IOException {
        byte[] data = { (byte) 0xFF, (byte) 0xFF, (byte) 0xFF, (byte) 0xFF, (byte) 0xFF, (byte) 0xFF, (byte) 0xFF, (byte) 0xFF, (byte) 0xFF };
        InputStream inputStream = new ByteArrayInputStream(data);
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readUE7(inputStream));
    }
}
